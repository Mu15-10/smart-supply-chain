name: Foundation CI

on:
  push:
  pull_request:

permissions:
  contents: read

jobs:
  backend:
    runs-on: ubuntu-latest
    services:
      postgres:
        image: postgres:16
        env:
          POSTGRES_USER: ci
          POSTGRES_PASSWORD: ci
          POSTGRES_DB: supply
        ports: ["5432:5432"]
        options: >-
          --health-cmd="pg_isready -U ci -d supply"
          --health-interval=5s --health-timeout=5s --health-retries=20
      redis:
        image: redis:7
        ports: ["6379:6379"]
    env:
      ENVIRONMENT: test
      JWT_SECRET: CHANGE_ME_CI_JWT_SECRET_12345678901234567890
      DATABASE_URL: postgresql+psycopg://ci:ci@localhost:5432/supply
      TEST_DATABASE_URL: postgresql+psycopg://ci:ci@localhost:5432/supply
      REDIS_URL: redis://localhost:6379/0
      ALLOWED_ORIGINS: http://localhost:5173
    steps:
      - uses: actions/checkout@11d5960a326750d5838078e36cf38b85af677262 # v4
      - uses: actions/setup-python@a26af69be951a213d495a4c3e4e4022e16d87065 # v5
        with:
          python-version: "3.12"
      - run: python -m pip install -e backend pytest ruff mypy pip-audit httpx2
      - run: ruff check backend
      - run: mypy backend/app --ignore-missing-imports
      - run: pytest -q backend
      - run: pip-audit
      - run: alembic upgrade head && alembic downgrade base && alembic upgrade head
        working-directory: backend

  frontend:
    runs-on: ubuntu-latest
    defaults:
      run:
        working-directory: dashboard
    steps:
      - uses: actions/checkout@11d5960a326750d5838078e36cf38b85af677262 # v4
      - uses: actions/setup-node@49933ea5288caeca8642d1e84afbd3f7d6820020 # v4
        with:
          node-version: 22
          cache: npm
          cache-dependency-path: dashboard/package-lock.json
      - run: npm ci
      - run: npm audit --omit=dev
      - run: npm run build

  secrets:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@11d5960a326750d5838078e36cf38b85af677262 # v4
        with:
          fetch-depth: 0
      - uses: gitleaks/gitleaks-action@ff98106e4c7b2bc287b24eaf42907196329070c7 # v2
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}

  container:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@11d5960a326750d5838078e36cf38b85af677262 # v4
      - uses: docker/setup-buildx-action@8d2750c68a42422c14e847fe6c8ac0403b4cbd6f # v3
      - run: docker build -t smart-supply-chain-backend:ci backend
      - uses: aquasecurity/trivy-action@ed142fd0673e97e23eac54620cfb913e5ce36c25 # v0.36.0
        with:
          image-ref: smart-supply-chain-backend:ci
          format: table
          exit-code: "1"
          ignore-unfixed: true
          severity: CRITICAL,HIGH
